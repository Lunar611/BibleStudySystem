
CREATE TABLE bible_translations (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    abbreviation VARCHAR(20) NOT NULL UNIQUE,
    language VARCHAR(50) NOT NULL,
    license TEXT,
    source_url TEXT
);

CREATE TABLE bible_books (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    testament VARCHAR(20) NOT NULL
        CHECK (testament IN ('OLD', 'NEW')),
    canonical_order INT NOT NULL UNIQUE
        CHECK (canonical_order > 0)
);

CREATE TABLE bible_chapters (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    translation_id BIGINT NOT NULL
        REFERENCES bible_translations(id),
    book_id BIGINT NOT NULL
        REFERENCES bible_books(id),
    chapter_number INT NOT NULL
        CHECK (chapter_number > 0),
    UNIQUE (translation_id, book_id, chapter_number)
);

CREATE TABLE bible_verses (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    chapter_id BIGINT NOT NULL
        REFERENCES bible_chapters(id) ON DELETE CASCADE,
    verse_number INT NOT NULL
        CHECK (verse_number > 0),
    verse_text TEXT NOT NULL,
    UNIQUE (chapter_id, verse_number)
);

CREATE INDEX idx_chapters_book
    ON bible_chapters(book_id, chapter_number);

CREATE INDEX idx_verses_chapter
    ON bible_verses(chapter_id, verse_number);
